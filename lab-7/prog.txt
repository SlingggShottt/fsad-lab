// ===== 1. Comparison Selectors =====

// $gt (Greater Than) & $lte (Less Than or Equal To)
// Goal: salary greater than 90,000 but less than or equal to 120,000
db.employees.find({
  salary: { $gt: 90000, $lte: 120000 }
})

// $in (Match Any Value in a List)
// Goal: employees in the HR or Sales department
db.employees.find({
  department: { $in: ["HR", "Sales"] }
})

// $ne (Not Equal)
// Goal: employees whose status is not "On Leave"
db.employees.find({
  status: { $ne: "On Leave" }
})

// ===== 2. Logical Selectors =====

// $and (Implicit)
// Goal: Engineering employees with a performance_score of 4.5 or higher
db.employees.find({
  department: "Engineering",
  performance_score: { $gte: 4.5 }
})

// $or
// Goal: employees in HR OR with a salary greater than 100,000
db.employees.find({
  $or: [
    { department: "HR" },
    { salary: { $gt: 100000 } }
  ]
})

// ===== 3. Combining Selectors for a Complex Query =====
// Goal: Active employees in Engineering or Sales, hired before 2024-01-01
db.employees.find({
  status: "Active",
  department: { $in: ["Engineering", "Sales"] },
  hire_date: { $lt: new Date("2024-01-01") }
})
